async function main() {
  const apiKey = process.env.ALCHEMYST_AI_API_KEY;

    console.error('ALCHEMYST_AI_API_KEY is required');

  const sessionB = `session_B-${runId}`;

    userId,
    sessionId: sessionA,
    query: 'vegan peanut'
  });

    m.content.toLowerCase().includes('vegan') && m.content.toLowerCase().includes('peanut')
  );

    await client.v1.context.memory.add({
      userId,
      sessionId: sessionA,
      content: 'User is vegan and allergic to peanuts'
    });
  }

  }
